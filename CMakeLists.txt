cmake_minimum_required(VERSION 3.22)
project(sdr_app VERSION 0.1.0 LANGUAGES C CXX)
include(cmake/MsvcDependencies.cmake)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(SDR_BUILD_GUI "Build the Qt Quick desktop receiver" ON)
option(SDR_ENABLE_SOAPY "Build optional SoapySDR device modules when SDK headers are installed" ON)
option(SDR_INSTALL_LINUX_PORTABLE "Install the Linux application with Qt runtime and SDK examples" OFF)
if(SDR_INSTALL_LINUX_PORTABLE AND (NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT SDR_BUILD_GUI))
  message(FATAL_ERROR "SDR_INSTALL_LINUX_PORTABLE requires the Linux GUI build")
endif()
include(CTest)
find_package(Threads REQUIRED)
add_subdirectory(core)
add_subdirectory(plugins)
if(SDR_BUILD_GUI)
  find_package(Qt6 6.10.0 EXACT REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Multimedia)
  qt_standard_project_setup(REQUIRES 6.10)
  add_subdirectory(app)
endif()
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
