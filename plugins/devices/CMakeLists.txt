if(NOT SDR_ENABLE_SOAPY)
  return()
endif()
find_path(SDR_SOAPY_INCLUDE_DIR SoapySDR/Device.h)
if(NOT SDR_SOAPY_INCLUDE_DIR)
  message(STATUS "Soapy SDK absent: optional device modules skipped; host/file/synthetic remain available")
  return()
endif()
add_library(sdr_device_soapy_backend STATIC soapy_backend.cpp)
set_target_properties(sdr_device_soapy_backend PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(sdr_device_soapy_backend PRIVATE ${SDR_SOAPY_INCLUDE_DIR} ../../sdk/include)
target_link_libraries(sdr_device_soapy_backend PRIVATE ${CMAKE_DL_LIBS})
foreach(family IN ITEMS hackrf rtlsdr airspy)
  set(target sdr_device_${family})
  add_library(${target} SHARED entry.cpp)
  target_include_directories(${target} PRIVATE ../../sdk/include)
  target_compile_definitions(${target} PRIVATE SDR_DEVICE_FAMILY="${family}" SDR_DEVICE_ID="org.sdr.device.${family}")
  target_link_libraries(${target} PRIVATE sdr_device_soapy_backend Threads::Threads)
  set_target_properties(${target} PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/devices" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/devices")
  file(GENERATE OUTPUT "$<TARGET_FILE_DIR:${target}>/${family}.json" CONTENT
    "{\"id\":\"org.sdr.device.${family}\",\"version\":\"0.1.0\",\"api_major\":1,\"api_minor\":0,\"kind\":\"device\",\"platform\":\"${CMAKE_SYSTEM_NAME}\",\"arch\":\"x86_64\",\"license\":\"MIT\",\"library\":\"$<TARGET_FILE_NAME:${target}>\"}")
endforeach()
