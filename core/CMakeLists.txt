add_library(sdr_core STATIC src/dsp.cpp src/source.cpp src/recording.cpp src/plugin.cpp)
target_include_directories(sdr_core PUBLIC include ../sdk/include)
target_link_libraries(sdr_core PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
set_target_properties(sdr_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(sdr_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(sdr_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(sdr_bench src/bench.cpp)
target_link_libraries(sdr_bench PRIVATE sdr_core)
if(SDR_ENABLE_SOAPY)
  # Use the public C ABI. Some third-party Soapy config packages inject global
  # MSVC-only flags even for a MinGW consumer, so discover the C SDK directly.
  find_path(SDR_SOAPY_INCLUDE_DIR SoapySDR/Device.h)
  if(SDR_SOAPY_INCLUDE_DIR)
    target_compile_definitions(sdr_core PRIVATE SDR_HAS_SOAPY=1)
    target_include_directories(sdr_core PRIVATE ${SDR_SOAPY_INCLUDE_DIR})
  else()
    message(STATUS "SoapySDR SDK absent: hardware RX unavailable; file and synthetic RX remain available")
  endif()
endif()
