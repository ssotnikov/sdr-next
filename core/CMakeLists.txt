add_library(sdr_core STATIC src/dsp.cpp src/source.cpp src/recording.cpp src/plugin.cpp src/device.cpp)
target_include_directories(sdr_core PUBLIC include ../sdk/include)
target_link_libraries(sdr_core PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
set_target_properties(sdr_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(sdr_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(sdr_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(sdr_bench src/bench.cpp)
target_link_libraries(sdr_bench PRIVATE sdr_core)
